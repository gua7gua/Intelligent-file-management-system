<script setup lang="ts">
import { computed, onMounted, ref } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import AnalysisItemDrawer from './components/AnalysisItemDrawer.vue'
import { createAnalysisTask, getAnalysisTasks, handleAnalysisItem } from '@/api/data-analysis'
import { getAnalysisTaskDetail } from '@/api/data-analysis'
import type { AnalysisItem, AnalysisTask, AnalysisTaskDetail, AnalysisTaskTypeValue } from '@/types/data-analysis'
import { AnalysisItemStatusLabel, AnalysisProblemTypeLabel, AnalysisTaskStatusLabel, AnalysisTaskTypeLabel } from '@/types/enums'

const newTaskType = ref<AnalysisTaskTypeValue>('mixed')

const filterStatus = ref<'' | 'running' | 'completed' | 'failed'>('')

const filteredTasks = computed(() => filterStatus.value ? tasks.value.filter((t) => t.status === filterStatus.value) : tasks.value)

async function load() {
  loading.value = true
  errorMsg.value = ''
  try {
    const page = await getAnalysisTasks({ pageSize: 50 })
    tasks.value = page.records
    if (tasks.value.length) await selectTask(tasks.value[0].id)
  } catch (e) {
    errorMsg.value = (e as Error).message || '研判任务加载失败'
  } finally {
    loading.value = false
  }
}

async function selectTask(id: number) {
  try {
    current.value = await getAnalysisTaskDetail(id)
    adoptedQueue.value = current.value.items.filter((i) => i.status === 'adopted')
  } catch (e) {
    ElMessage.warning((e as Error).message)
  }
}

function openItem(item: AnalysisItem) {
  selectedItem.value = item
  drawerVisible.value = true
}

async function onHandle(action: 'adopted' | 'rejected', note: string) {
  if (!selectedItem.value) return
  try {
    await handleAnalysisItem(selectedItem.value.id, { action, note })
    ElMessage.success(action === 'adopted' ? '已采纳，加入待处理队列' : '已不采纳')
    drawerVisible.value = false
    if (current.value) await selectTask(current.value.id)
  } catch (e) {
    ElMessage.warning((e as Error).message)
  }
}

async function startScan() {
  try {
    await ElMessageBox.confirm('将创建研判任务并开始扫描，确认继续？', '开始扫描', { type: 'info' })
  } catch { return }
  try {
    const detail = await createAnalysisTask({
      taskType: newTaskType.value,
      rule: { categoryIds: [1, 3], formedYearStart: 2010, formedYearEnd: 2026, includeAiSuggestion: newIncludeAi.value },
    })
    ElMessage.success('扫描已开始')
    await load()
    await selectTask(detail.id)
  } catch (e) {
    ElMessage.warning((e as Error).message)
  }
}

function generateList() {
  if (!adoptedQueue.value.length) {
    ElMessage.warning('请先采纳建议，再生成待补充清单')
    return
  }
  ElMessage.success(`已生成待补充清单（${adoptedQueue.value.length} 条），请前往档案管理页人工处理`)
}

onMounted(load)

function collectData(): CompilationWriteData {
  if (editorRef.value) contentHtml.value = editorRef.value.innerHTML
  return {
    title: title.value, compilationType: compilationType.value, dateRangeText: dateRangeText.value,
    keywords: keywords.value, summary: summary.value, contentHtml: contentHtml.value,
    materialArchiveIds: selectedMaterials.value.map((m) => m.id),
  }
}

async function saveDraft() {
  if (!title.value.trim()) { ElMessage.warning('请填写标题'); return }
  saving.value = true
  try {
    if (props.detail) await updateCompilation(props.detail.id, collectData())
    else { const d = await createCompilation(collectData()); emit('changed'); ElMessage.success('草稿已保存'); return }
    ElMessage.success('草稿已保存'); emit('changed')
  } catch (e) { ElMessage.warning((e as Error).message) } finally { saving.value = false }
}

async function generateBody() {
  if (!props.detail) { ElMessage.warning('请先保存草稿'); return }
  generating.value = true
  try {
    await generateCompilationBody(props.detail.id)
    ElMessage.success('正文文件已生成'); emit('changed')
  } catch (e) { ElMessage.warning((e as Error).message) } finally { generating.value = false }
}

async function archive() {
  if (!props.detail) return
  if (props.detail.status !== 'generated') { ElMessage.warning('请先生成正文文件'); return }
  if (!props.detail.summary?.trim()) { ElMessage.warning('请填写摘要'); return }
  if ((props.detail.materialCount ?? 0) === 0) { ElMessage.warning('请添加至少一个素材'); return }
  try {
    await ElMessageBox.confirm('确认纯电子入库？入库后生成正式档号且不可再编辑。', '确认入库', { type: 'warning' })
  } catch { return }
  try {
    const archived = await archiveCompilation(props.detail.id, { fondsId: 1, categoryId: 1, formedDate: '2026-06-17', retentionPeriod: 'permanent', openStatus: 'open' })
    ElMessage.success(`已入库，正式档号 ${archived.archiveNo}`); emit('changed')
  } catch (e) { ElMessage.warning((e as Error).message) }
}

async function search() {
  const page = await searchMaterials({ keyword: searchKeyword.value, pageSize: 50 })
  candidates.value = page.records
}

function addMaterial(m: MaterialCandidate) {
  if (!selectedMaterials.value.some((x) => x.id === m.id)) selectedMaterials.value.push(m)
}
function removeMaterial(id: number) {
  selectedMaterials.value = selectedMaterials.value.filter((m) => m.id !== id)
}

<script setup lang="ts">
import { computed, onMounted, ref } from 'vue'

  return true
}))

  archivedCount: list.value.filter((c) => c.status === 'archived').length,

}))

  }

  if (currentDetail.value) {
    try { currentDetail.value = await getCompilationDetail(currentDetail.value.id) } catch { /* 新建的草稿无 id，忽略 */ }
  }
}

onMounted(load)
